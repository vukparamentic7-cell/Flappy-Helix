// CRTANJE NAPREDNIJIH I LEPŠIH STUBOVA for (let pipe of pipes) { // 1. Glavni gradijent cevi (svetlije sa leve, tamnije sa desne strane za 3D efekat) let pipeGradient = ctx.createLinearGradient(pipe.x, 0, pipe.x + pipe.width, 0); pipeGradient.addColorStop(0, "#8ED142"); // Sjajno zelena pipeGradient.addColorStop(0.3, "#73BF2E"); pipeGradient.addColorStop(0.7, "#558022"); pipeGradient.addColorStop(1, "#364D14"); // Duboka senka ctx.fillStyle = pipeGradient; // --- GORNJI STUB --- ctx.fillRect(pipe.x, 0, pipe.width, pipe.top); // Kapa gornjeg stuba (širi deo na kraju) ctx.fillRect(pipe.x - 4, pipe.top - 24, pipe.width + 8, 24); // --- DONJI STUB --- ctx.fillRect(pipe.x, canvas.height - pipe.bottom, pipe.width, pipe.bottom); // Kapa donjeg stuba (širi deo na početku) ctx.fillRect(pipe.x - 4, canvas.height - pipe.bottom, pipe.width + 8, 24); // 2. Svetlosni odsjaj (efekat sjaja na cevi - tanka svetla linija) ctx.fillStyle = "rgba(255, 255, 255, 0.25)"; ctx.fillRect(pipe.x + 4, 0, 4, pipe.top - 24); ctx.fillRect(pipe.x + 4, canvas.height - pipe.bottom + 24, 4, pipe.bottom - 24); // 3. Spoljašnje oivičenje za čist i definisan retro-moderni izgled ctx.strokeStyle = "#1A260A"; ctx.lineWidth = 2; // Okvir gornjeg stuba i kape ctx.strokeRect(pipe.x, 0, pipe.width, pipe.top); ctx.strokeRect(pipe.x - 4, pipe.top - 24, pipe.width + 8, 24); // Okvir donjeg stuba i kape ctx.strokeRect(pipe.x, canvas.height - pipe.bottom, pipe.width, pipe.bottom); ctx.strokeRect(pipe.x - 4, canvas.height - pipe.bottom, pipe.width + 8, 24); }
