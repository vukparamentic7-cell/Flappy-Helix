// Primer funkcije za crtanje realističnijih stubova (cevi) pipes.forEach(pipe => { // Definisanje gradijenta za stub da dobije 3D efekat (svetlije levo, tamnije desno) let pipeGradient = ctx.createLinearGradient(pipe.x, 0, pipe.x + pipe.width, 0); pipeGradient.addColorStop(0, "#73bf2e"); // Svetlija zelena pipeGradient.addColorStop(0.5, "#558022"); // Srednja zelena pipeGradient.addColorStop(1, "#3b5917"); // Tamnija zelena ctx.fillStyle = pipeGradient; // --- GORNJI STUB --- ctx.fillRect(pipe.x, 0, pipe.width, pipe.top); // Kapa gornjeg stuba (ivica) ctx.fillRect(pipe.x - 3, pipe.top - 20, pipe.width + 6, 20); // --- DONJI STUB --- ctx.fillRect(pipe.x, canvas.height - pipe.bottom, pipe.width, pipe.bottom); // Kapa donjeg stuba (ivica) ctx.fillRect(pipe.x - 3, canvas.height - pipe.bottom, pipe.width + 6, 20); // Dodavanje tanke ivice (obrub) oko cevi za oštriji izgled ctx.strokeStyle = "#22330d"; ctx.lineWidth = 2; ctx.strokeRect(pipe.x, 0, pipe.width, pipe.top); ctx.strokeRect(pipe.x - 3, pipe.top - 20, pipe.width + 6, 20); ctx.strokeRect(pipe.x, canvas.height - pipe.bottom, pipe.width, pipe.bottom); ctx.strokeRect(pipe.x - 3, canvas.height - pipe.bottom, pipe.width + 6, 20); });
